{"market_stage":{"phase":"弱修复失败","detail":"全天弱势，资金切换防御"},"direction_priority":[{"direction":"燃气轮机","intensity":"🔥🔥🔥"},{"direction":"机器人","intensity":"🔥🔥"}],"position_stance":"空仓等待","active_opportunities":[{"stock":"万泽股份","code":"000534","pattern":"技术支撑","status":"未触发","ratio":"3:1"}],"intraday_narrative":[{"time":"15:20","summary":"全天弱势，明日继续观察，等企稳信号"}]}
